50. Pow(x, n)

### Problem
Implement pow(x, n), which calculates x raised to the power n (i.e., xn).

```
Example 1:
Input: x = 2.10000, n = 3
Output: 9.26100
```

### Solution 1
```c =
double myPow(double x, int n) {
    int i;
    double result=x;
    if(n==0){
        return 1;
    }

    if(n<0){
        n=-n+2;
        for(i=0;i<n-1;i++){
            result = result/x;
        }
    }
    else{
        for(i=0;i<n-1;i++){
            result = result*x;
        }
    }

    return result;
}
```

### Solution 2
```c =
double myPow(double x, int n) {

    long long N = n; // prevent overflow
    if(N<0){
        x = 1.0 / x;
        N = -N;
    }

    double result = 1.0;
    double cur_pow = x;

    while(N>0){
        if(N&1){
            result *= cur_pow;
        }
        
        cur_pow *= cur_pow;
        N >>= 1;            // n/2
    }

    return result;
}
```
